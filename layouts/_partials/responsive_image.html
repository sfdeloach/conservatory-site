{{/* Note: This partial requires a dictionary to be passed to it that defines. All fields are required unless noted: context resource (file name of image, colocated with content) sizes (https://developer.mozilla.org/en-US/docs/Web/API/HTMLImageElement/sizes) aspectRatio (optional, if not specified, image is resized, 4:3 = 1.333, 2:3 = 0.667) widths (slice of widths to process) cssClass cropAnchor (optional, not used if image is resized) altText (optional) isLazyLoaded (optional, defaults to 'eager') hasFetchPriority (optional, defaults to 'auto') The build fails if no image in the page bundle matches resource. If the image is smaller than every requested width, a single WebP at its own size is used. */}} {{- with .context }} {{- with $src := .Resources.GetMatch $.resource }} {{- $srcSet := slice }} {{- $images := slice }} {{- range $.widths }} {{- $spec := "" }} {{- if $.aspectRatio }} {{- $h := int (math.Round (div . (float $.aspectRatio))) }} {{- if and (le . $src.Width) (le $h $src.Height) }} {{- $spec = printf "fill %dx%d %s webp q80" . $h $.cropAnchor }} {{- end }} {{- else if le . $src.Width }} {{- $spec = printf "resize %dx webp q80" . }} {{- end }} {{- with $spec }} {{- $img := $src.Process . }} {{- $images = $images | append $img }} {{- $srcSet = $srcSet | append (printf "%s %dw" $img.RelPermalink $img.Width) }} {{- end }} {{- end }} {{- if not $images }} {{- $spec := printf "resize %dx webp q80" $src.Width }} {{- if $.aspectRatio }} {{- $w := int (math.Min $src.Width (math.Floor (mul $src.Height (float $.aspectRatio)))) }} {{- $h := int (math.Round (div $w (float $.aspectRatio))) }} {{- $spec = printf "fill %dx%d %s webp q80" $w $h $.cropAnchor }} {{- end }} {{- $img := $src.Process $spec }} {{- $images = slice $img }} {{- $srcSet = slice (printf "%s %dw" $img.RelPermalink $img.Width) }} {{- end }} {{- with $images -}} {{- $fallback := index . 0 }} {{ end -}} {{- else }} {{- errorf "responsive_image: no image matching %v in %s" $.resource .Path }} {{ end -}} {{ end -}}
