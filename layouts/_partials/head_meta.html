{{- /* Search and sharing metadata for the page's . Called from baseof.html with the page as its context. - Description: the front matter `description`; otherwise the page summary as plain text, cut to about 160 characters; otherwise the homepage's `description`. A page that ends with none warns, which fails `make build`. - Canonical link and og:url: the page's own permalink. - Robots: `noindex` whenever the base URL's host is not `params.productionHost`, so the staging copy on GitHub Pages and local builds stay out of search engines. At cutover, CI builds with the production domain and the tag disappears on its own. - Sharing image: the homepage's hero, a faculty bio's headshot (square, with a `summary` card), or the page's `params.image` (cropped to 1200x630, with a `summary_large_image` card). A page without one uses the homepage hero. The image is a JPEG, which every link-preview service reads, and its URL is absolute. */ -}} {{- $productionHost := site.Params.productionHost }} {{- if not $productionHost }} {{- errorf "head_meta.html: params.productionHost is not set in hugo.toml" }} {{- end }} {{- if ne (urls.Parse site.BaseURL).Host $productionHost }} {{- end }} {{- /* Description */}} {{- $description := .Description }} {{- if not $description }} {{- $description = .Summary | plainify | htmlUnescape }} {{- $description = replaceRE `\s+` " " $description | strings.TrimSpace | truncate 160 }} {{- end }} {{- if not $description }} {{- $description = site.Home.Description }} {{- end }} {{- if not $description }} {{- warnf "head_meta.html: %s has no description, and the homepage has none to fall back on" .Path }} {{- end }} {{- /* Sharing image: a source resource, how to crop it, and its alt text */}} {{- $image := "" }} {{- $square := false }} {{- $imageAlt := "" }} {{- $hero := site.Home.Params.hero }} {{- if .IsHome }} {{- $image = .Resources.Get $hero.image }} {{- $imageAlt = $hero.imageAlt }} {{- else if eq .Layout "page_faculty_bio" }} {{- $image = .Resources.GetMatch "headshot.*" }} {{- $square = true }} {{- else }} {{- with .Params.image }} {{- $image = $.Resources.Get . }} {{- end }} {{- end }} {{- if not $image }} {{- $image = site.Home.Resources.Get $hero.image }} {{- $imageAlt = $hero.imageAlt }} {{- end }} {{- if not $image }} {{- errorf "head_meta.html: no sharing image for %s, and the homepage hero image was not found" .Path }} {{- end }} {{- $size := "1200x630" }} {{- if $square }} {{- $side := math.Min 600 $image.Width $image.Height }} {{- $size = printf "%dx%d" (int $side) (int $side) }} {{- end }} {{- $image = $image.Fill (printf "%s Center q80 jpg" $size) }} {{- $title := .Title }} {{- if .IsHome }} {{- $title = site.Title }} {{- end }} {{- with $imageAlt }} {{- end }}
