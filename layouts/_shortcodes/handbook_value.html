An unknown path fails the build. */}} {{- $path := .Get 0 }} {{- $v := (site.GetPage "/handbook").Params }} {{- range split $path "." }} {{- if reflect.IsMap $v }} {{- $v = index $v (lower .) }} {{- else }} {{- $v = "" }} {{- end }} {{- end }} {{- if not $v }} {{- errorf "handbook_value: %q not found in content/handbook/index.md params (%s)" $path .Position }} {{- end }} {{- $v -}}
